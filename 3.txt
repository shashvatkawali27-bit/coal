Program 1: HEX to BCD
.MODEL SMALL
.STACK 100H
.DATA
HEXNUM DB 0AH
BCD DB 3 DUP(0)
.CODE
MAIN PROC
MOV AX, @DATA
MOV DS, AX
; Load hexadecimal number
MOV AL, HEXNUM
MOV AH, 00H
; Divide by 100
MOV BL, 100
DIV BL
; Store hundreds digit
MOV BCD, AL
; Remainder is in AH
MOV AL, AH
MOV AH, 00H
; Divide remainder by 10
MOV BL, 10
DIV BL
; Store tens and units
MOV BCD+1, AL
MOV BCD+2, AH
; Display hundreds digit
MOV DL, BCD

ADD DL, 30H
MOV AH, 02H
INT 21H
; Display tens digit
MOV DL, BCD+1
ADD DL, 30H
MOV AH, 02H
INT 21H
; Display units digit
MOV DL, BCD+2
ADD DL, 30H
MOV AH, 02H
INT 21H
; Exit
MOV AH, 4CH
INT 21H
MAIN ENDP
END MAIN



Program 2: ASCII to BCD
ASSUME CS:CODE,DS:DATA,SS:STACK
STACK SEGMENT STACK
DW 100 DUP(?)
STACK ENDS
DATA SEGMENT
ASCII DB &#39;5&#39;
MSG DB &#39;BCD = $&#39;
DATA ENDS
CODE SEGMENT

START:
MOV AX,DATA
MOV DS,AX
; Display message
LEA DX,MSG
MOV AH,09H
INT 21H
; ASCII to BCD
MOV AL,ASCII
SUB AL,30H
; Convert back to ASCII for display
ADD AL,30H
MOV DL,AL
MOV AH,02H
INT 21H
; Exit
MOV AH,4CH
INT 21H

CODE ENDS
END START




Program 3: BCD to ASCII
ASSUME CS:CODE,DS:DATA,SS:STACK
STACK SEGMENT STACK

DW 100 DUP(?)
STACK ENDS
DATA SEGMENT
BCD DB 05H
MSG DB &#39;ASCII = $&#39;
DATA ENDS
CODE SEGMENT
START:
MOV AX,DATA
MOV DS,AX
; Display message
LEA DX,MSG
MOV AH,09H
INT 21H
; BCD to ASCII
MOV AL,BCD
ADD AL,30H
; Display ASCII character
MOV DL,AL
MOV AH,02H
INT 21H
; Exit
MOV AH,4CH
INT 21H
CODE ENDS
END START
