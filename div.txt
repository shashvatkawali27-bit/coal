
.MODEL SMALL
.STACK 100H

.DATA
    DIVIDEND  DW 100
    DIVISOR   DW 7
    QUOTIENT  DW ?
    REMAINDER DW ?
    MSGQ      DB 13,10,'Quotient = $'
    MSGR      DB 13,10,'Remainder = $'

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX

    MOV AX, DIVIDEND
    MOV BX, DIVISOR
    XOR DX, DX
    MOV CX, 16

RESTORE_LOOP:
    SHL AX, 1
    RCL DX, 1
    SUB DX, BX
    JC RESTORE
    OR AX, 0001H
    JMP NEXT_BIT

RESTORE:
    ADD DX, BX

NEXT_BIT:
    LOOP RESTORE_LOOP

    MOV QUOTIENT, AX
    MOV REMAINDER, DX

    LEA DX, MSGQ
    MOV AH, 09H
    INT 21H
    MOV AX, QUOTIENT
    CALL DISPLAY_NUMBER

    LEA DX, MSGR
    MOV AH, 09H
    INT 21H
    MOV AX, REMAINDER
    CALL DISPLAY_NUMBER

    MOV AH, 4CH
    INT 21H
MAIN ENDP

DISPLAY_NUMBER PROC
    MOV BX, 10
    XOR CX, CX

CONVERT:
    XOR DX, DX
    DIV BX
    PUSH DX
    INC CX
    CMP AX, 0
    JNE CONVERT

PRINT:
    POP DX
    ADD DL, '0'
    MOV AH, 02H
    INT 21H
    LOOP PRINT
    RET
DISPLAY_NUMBER ENDP

END MAIN
